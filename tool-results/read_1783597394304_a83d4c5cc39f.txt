    78→  function reset() {

   259→function CandidateCard({
   260→  candidate,
   261→  onOpen,
   262→}: {
   263→  candidate: CandidateDTO
   264→  onOpen: (id: string) => void
   265→}) {
   266→  return (

   283→              {initials(candidate.name)}
   284→            </AvatarFallback>
   285→          </Avatar>
   286→          <div className="flex-1 min-w-0">
   287→            <div className="font-medium truncate">{candidate.name}</div>
   288→            <div className="text-xs text-muted-foreground truncate">{candidate.headline}</div>
   289→            <div className="text-[11px] text-muted-foreground truncate">{candidate.location}</div>
   290→          </div>
   291→          <CandidateStatusBadge status={candidate.status} />
   292→        </div>
   293→
   294→        {candidate.tags.length > 0 && (
   295→          <div className="flex flex-wrap gap-1">
   296→            {candidate.tags.slice(0, 4).map((t) => (
   297→              <Badge key={t} variant="outline" className="text-[10px]">
   298→                {t}
   299→              </Badge>
   300→            ))}
   301→          </div>
   302→        )}
   303→
   304→        {candidate.tools.length > 0 && (
   305→          <div className="flex flex-wrap gap-1">
   306→            {candidate.tools.slice(0, 3).map((t) => (
   307→              <span
   308→                key={t}
   309→                className="rounded bg-primary/10 px-1.5 py-0.5 text-[10px] text-primary"
   310→              >
   311→                {t}
   312→              </span>
   313→            ))}
   314→          </div>
   315→        )}
   316→
   317→        <Button variant="ghost" size="sm" className="w-full" onClick={(e) => {
   318→          e.stopPropagation()
   319→          onOpen(candidate.id)
   320→        }}>
   321→          View profile
   322→        </Button>
   323→      </CardContent>
   324→    </Card>
   325→  )

   718→export function CandidatesView() {
   719→  const { data: candidates, isLoading } = useCandidates()
   720→  const [search, setSearch] = useState('')
   721→  const [selectedId, setSelectedId] = useState<string | null>(null)
   722→  const [sheetOpen, setSheetOpen] = useState(false)
   723→
   724→  const filtered = (candidates ?? []).filter((c) => {
   725→    if (!search.trim()) return true
   726→    const q = search.toLowerCase()
   727→    return (
   728→      c.name.toLowerCase().includes(q) ||
   729→      c.headline.toLowerCase().includes(q) ||
   730→      c.location.toLowerCase().includes(q) ||
   731→      c.tools.some((t) => t.toLowerCase().includes(q)) ||
   732→      c.tags.some((t) => t.toLowerCase().includes(q))
   733→    )
   734→  })
   735→
   736→  const selected = filtered.find((c) => c.id === selectedId) ?? null
   737→
   738→  function open(id: string) {
   739→    setSelectedId(id)
   740→    setSheetOpen(true)
   741→  }
   742→
   743→  return (
   744→    <div className="space-y-4">
   745→      <Card>
   746→        <CardHeader className="flex-row items-center justify-between gap-2">
   747→          <div>
   748→            <CardTitle className="flex items-center gap-2">
   749→              <Users className="size-4 text-primary" />
   750→              Candidate database
   751→            </CardTitle>
   752→            <CardDescription className="mt-1">
   753→              Structured, outcome-based profiles — the foundation the matcher runs on.
   754→            </CardDescription>
   755→          </div>
   756→          <AddCandidateDialog />
   757→        </CardHeader>
   758→        <CardContent>
   759→          <div className="relative mb-4">
   760→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   761→            <Input
   762→              value={search}
   763→              onChange={(e) => setSearch(e.target.value)}
   764→              placeholder="Search by name, headline, tool, location, or tag…"
   765→              className="pl-9"
   766→              aria-label="Search candidates"
   767→            />
   768→          </div>
   769→
   770→          {isLoading ? (
   771→            <CandidateGridSkeleton />
   772→          ) : filtered.length === 0 ? (
   773→            search.trim() ? (
   774→              <div className="py-8 text-center text-sm text-muted-foreground">
   775→                No candidates match "{search}".
   776→              </div>
   777→            ) : (
   778→              <CandidateEmpty />
   779→            )
   780→          ) : (
   781→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   782→              {filtered.map((c) => (
   783→                <CandidateCard key={c.id} candidate={c} onOpen={open} />
   784→              ))}
   785→            </div>
   786→          )}
   787→        </CardContent>
   788→      </Card>
   789→
   790→      <CandidateDetail
   791→        candidate={selected}
   792→        open={sheetOpen}
   793→        onOpenChange={setSheetOpen}
   794→      />
   795→    </div>
   796→  )
   797→}
